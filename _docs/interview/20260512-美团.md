class Solution {
    public ListNode reverseBetween(ListNode head, int left, int right) {
        // write your code here.
        ListNode begin = head;
        for(int i = 0; i < left - 1; i++) {
          begin = begin.next;
        }
        ListNode end = head;
        for(int i = 0; i < right - 1; i++) {
          end = end.next;
        }
        ListNode tempBack = end.next;
        end.next = null;
        ListNode mid = reverse(begin);
        ListNode targetTail = mid;
        while(targetTail != null && targetTail.next == null) {
          targetTail = targetTail.next;
        }
        begin.next = mid;
        targetTail.next = end;
        return head;
        
    }
    
    private ListNode reverse(ListNode node) {
      if(node == null || node.next == null) {
        return node;
      }
      ListNode prev = null;
      ListNode cur = node;
      while(cur != null) {
        ListNode temp = cur;
        cur.next = prev;
        prev = cur;
        cur = cur.next;
      }
      return prev;
    }
}
